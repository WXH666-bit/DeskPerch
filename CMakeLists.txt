cmake_minimum_required(VERSION 3.24)
project(DeskPerch VERSION 1.1.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(deskperch_core src/model.cpp src/settings.cpp src/devices.cpp src/hidpp.cpp src/langtu.cpp)
target_include_directories(deskperch_core PUBLIC src)
target_compile_definitions(deskperch_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(deskperch_core PUBLIC /utf-8 /W4 /permissive- /EHsc)
target_link_libraries(deskperch_core PUBLIC setupapi cfgmgr32 hid BluetoothApis user32 advapi32 shell32 ole32)
add_executable(DeskPerch WIN32 src/main.cpp src/desktop.cpp src/render.cpp src/app.rc)
target_link_libraries(DeskPerch PRIVATE deskperch_core gdiplus wtsapi32 dwmapi shcore)
target_link_options(DeskPerch PRIVATE /MANIFEST:NO)
add_executable(DeskPerchProbe tools/probe.cpp src/render.cpp src/desktop.cpp)
target_link_libraries(DeskPerchProbe PRIVATE deskperch_core gdiplus shcore)
enable_testing()
add_executable(DeskPerchTests tests/tests.cpp)
target_link_libraries(DeskPerchTests PRIVATE deskperch_core)
add_test(NAME core COMMAND DeskPerchTests)
install(TARGETS DeskPerch RUNTIME DESTINATION .)

add_executable(DeskPerchLayoutTests tests/layout.cpp src/render.cpp)
target_link_libraries(DeskPerchLayoutTests PRIVATE deskperch_core gdiplus)
add_test(NAME layout COMMAND DeskPerchLayoutTests)
